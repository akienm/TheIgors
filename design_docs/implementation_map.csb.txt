[DATA_FLOW]

FLOW|user_message (REPL)
  stdin → _stdin_reader → stdin_queue → Igor.run() → Igor._process()
  → thalamus.process() → [preparse(Ollama)] → [habit_match || pfc.reason(Anthropic||Ollama)]
  → console.print(response) → cortex.store(EPISODIC) → cortex.write_ring() → dashboard.render()
  → web_server.update_stats()

FLOW|user_message (web)
  browser → WS /ws → web/server.py _ws_endpoint → web_server.incoming.put()
  → network/listener.py _poll_web() → net_listener.incoming.put(NetworkMessage source=web)
  → Igor._drain_network() → Igor._process(synthetic) → web_server.send(response)
  → _broadcast() → all WebSocket clients

FLOW|user_message (discord)
  Discord API → discord_bot IgorBot.on_message() → discord_bot.incoming.put()
  → listener._poll_discord() → net_listener.incoming → Igor._drain_network()
  → Igor._process() [no web_server.send for discord; uses discord_bot.send() via tool]

FLOW|TWM_cycle
  push_sources|interruptors → cortex.twm_push()
  → NarrativeEngine.run() [background thread, every 30s-5min]
  → twm_read() → Ollama call → _apply_output()
  → twm_mark_integrated() + twm_update_salience() + cortex.store(Memory) [if importance≥0.7]
  → cortex.write_ring(NE narrative fragment)

FLOW|self_edit
  Igor detects edit intent → AnthropicReasoner auto-switches to Haiku
  → read_source_file() [reads current state] → patch_source_file(old,new,reason)
  → ast.parse() [syntax check] → write file → _git_commit_and_push() → restart required
